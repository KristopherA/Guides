INSTALL
  Ubuntu:  sudo apt install -y tcpdump wireshark tshark
  macOS:   brew install wireshark  (or use built-in tcpdump)

PERMISSIONS
  # Allow non-root to capture on Ubuntu (then log out/in):
  sudo usermod -aG wireshark $USER
  sudo dpkg-reconfigure wireshark-common   # select "Yes"

  # Or just run with sudo:
  sudo tcpdump ...

================================================================================
SECTION 1: TCPDUMP QUICK REFERENCE
================================================================================

--------------------------------------------------------------------------------
1.1 INTERFACE SELECTION
--------------------------------------------------------------------------------

LIST AVAILABLE INTERFACES
  sudo tcpdump -D
  ip link show   # alternative

CAPTURE ON ALL INTERFACES
  sudo tcpdump -i any

CAPTURE ON SPECIFIC INTERFACE
  sudo tcpdump -i ens3
  sudo tcpdump -i lo         # loopback (local traffic)

--------------------------------------------------------------------------------
1.2 ESSENTIAL FLAGS
--------------------------------------------------------------------------------

  -i <iface>     interface (any = all)
  -n             don't resolve hostnames (faster, cleaner)
  -nn            don't resolve hostnames OR port names
  -v             verbose (more detail)
  -vv            more verbose
  -c <N>         capture only N packets then exit
  -w <file>      write to pcap file (open in Wireshark later)
  -r <file>      read from pcap file
  -s 0           capture full packet (default truncates at 96 bytes)
  -X             show hex + ASCII payload
  -A             show ASCII payload only
  -q             quiet (minimal output)
  -e             show Ethernet header (MAC addresses)
  -t             no timestamps
  -S             absolute sequence numbers

--------------------------------------------------------------------------------
1.3 CAPTURE FILTERS (BPF syntax)
--------------------------------------------------------------------------------

BY HOST
  sudo tcpdump -i any -nn host 10.0.1.10
  sudo tcpdump -i any -nn src host 10.0.1.10       # only from this IP
  sudo tcpdump -i any -nn dst host 10.0.1.10       # only to this IP

BY PORT
  sudo tcpdump -i any -nn port 443
  sudo tcpdump -i any -nn src port 80
  sudo tcpdump -i any -nn dst port 22
  sudo tcpdump -i any -nn portrange 8000-9000

BY PROTOCOL
  sudo tcpdump -i any -nn tcp
  sudo tcpdump -i any -nn udp
  sudo tcpdump -i any -nn icmp
  sudo tcpdump -i any -nn arp

COMBINE FILTERS (and/or/not)
  sudo tcpdump -i any -nn host 10.0.1.10 and port 443
  sudo tcpdump -i any -nn port 80 or port 443
  sudo tcpdump -i any -nn not port 22                  # exclude SSH
  sudo tcpdump -i any -nn host 10.0.1.10 and not port 22

BY TCP FLAGS
  sudo tcpdump -i any -nn 'tcp[tcpflags] & tcp-syn != 0'    # SYN packets (new connections)
  sudo tcpdump -i any -nn 'tcp[tcpflags] & tcp-rst != 0'    # RST packets (connection resets)
  sudo tcpdump -i any -nn 'tcp[tcpflags] == 0x02'           # SYN only (no SYN-ACK)

BY NETWORK
  sudo tcpdump -i any -nn net 10.0.0.0/8
  sudo tcpdump -i any -nn src net 192.168.0.0/16

BY PACKET SIZE
  sudo tcpdump -i any -nn greater 1000     # packets larger than 1000 bytes
  sudo tcpdump -i any -nn less 64          # tiny packets (possible issue)

MULTICAST/BROADCAST
  sudo tcpdump -i any -nn ether broadcast
  sudo tcpdump -i any -nn ether multicast

--------------------------------------------------------------------------------
1.4 COMMON DIAGNOSTIC SCENARIOS
--------------------------------------------------------------------------------

WATCH ALL CONNECTIONS TO A WEB SERVER
  sudo tcpdump -i ens3 -nn -c 100 port 80 or port 443

CHECK IF A PORT IS BEING REACHED
  sudo tcpdump -i any -nn -c 20 dst port 5432
  # If no packets appear: traffic never arrives (firewall/routing issue)
  # If packets appear: traffic arrives but app may not respond

DIAGNOSE A REFUSED CONNECTION (see RST packets)
  sudo tcpdump -i any -nn 'tcp[tcpflags] & tcp-rst != 0 and host 10.0.1.10'

WATCH SSH CONNECTIONS (who is connecting)
  sudo tcpdump -i ens3 -nn -c 50 port 22

WATCH DNS QUERIES
  sudo tcpdump -i any -nn -A port 53 -c 20

CAPTURE DHCP TRAFFIC
  sudo tcpdump -i ens3 -nn port 67 or port 68 -vv

WATCH ICMP (ping/traceroute)
  sudo tcpdump -i any -nn icmp

CHECK ARP (find who has an IP / detect IP conflicts)
  sudo tcpdump -i ens3 -nn arp
  sudo arping -I ens3 10.0.1.10 -c 3   # who responds to this IP?

CAPTURE NTP TRAFFIC
  sudo tcpdump -i any -nn port 123

CAPTURE LDAP (AD authentication traffic)
  sudo tcpdump -i any -nn port 389 or port 636 -A | grep -i "uid\|cn\|samaccountname"

CAPTURE KERBEROS TRAFFIC
  sudo tcpdump -i any -nn port 88

WATCH FOR BANDWIDTH HOG (by packet size, show large packets)
  sudo tcpdump -i ens3 -nn -c 200 greater 1400 | awk '{print $3}' | cut -d. -f1-4 | sort | uniq -c | sort -rn | head -10

--------------------------------------------------------------------------------
1.5 SAVING AND READING CAPTURES
--------------------------------------------------------------------------------

SAVE CAPTURE TO FILE (open later in Wireshark)
  sudo tcpdump -i ens3 -nn -w /tmp/capture-$(date +%Y%m%d-%H%M).pcap
  # Ctrl+C to stop

SAVE WITH ROTATION (100MB max file size, keep 5 files)
  sudo tcpdump -i ens3 -nn -C 100 -W 5 -w /tmp/capture.pcap

READ A SAVED CAPTURE
  tcpdump -r /tmp/capture.pcap
  tcpdump -r /tmp/capture.pcap -nn host 10.0.1.10

COPY PCAP TO LOCAL MACHINE FOR WIRESHARK ANALYSIS
  scp ubuntu@server:/tmp/capture.pcap ./


================================================================================
SECTION 2: TSHARK QUICK REFERENCE
================================================================================

tshark = Wireshark's CLI tool. More powerful filters and display than tcpdump.
Install: sudo apt install -y tshark

--------------------------------------------------------------------------------
2.1 BASIC TSHARK USAGE
--------------------------------------------------------------------------------

LIVE CAPTURE (similar to tcpdump)
  sudo tshark -i ens3

CAPTURE WITH DISPLAY FILTER
  sudo tshark -i ens3 -Y "http"
  sudo tshark -i ens3 -Y "dns"
  sudo tshark -i ens3 -Y "tcp.flags.reset == 1"
  sudo tshark -i any -Y "ip.addr == 10.0.1.10 and tcp.port == 443"

SHOW SPECIFIC FIELDS ONLY
  sudo tshark -i ens3 -Y "dns" -T fields -e frame.time -e ip.src -e dns.qry.name
  sudo tshark -i ens3 -Y "http.request" -T fields -e ip.src -e http.host -e http.request.uri

CAPTURE TO FILE AND READ
  sudo tshark -i ens3 -w /tmp/capture.pcap
  tshark -r /tmp/capture.pcap -Y "tcp.analysis.retransmission"

--------------------------------------------------------------------------------
2.2 TSHARK STATISTICS
--------------------------------------------------------------------------------

SHOW TOP TALKERS BY IP
  sudo tshark -i ens3 -q -z ip_hosts,tree -c 5000

SHOW ALL CONVERSATIONS (IP pairs)
  sudo tshark -i ens3 -q -z conv,tcp -c 5000 | head -30

PROTOCOL HIERARCHY
  sudo tshark -i ens3 -q -z io,phs -c 5000

SHOW HTTP REQUESTS
  sudo tshark -i ens3 -Y "http.request" -T fields -e ip.src -e ip.dst -e http.host -e http.request.uri -E separator=, | head -20

SECTION 3: WIRESHARK DISPLAY FILTERS (GUI)
================================================================================

These filters go in the Wireshark filter bar (not capture filters).

FILTER BY IP
  ip.addr == 10.0.1.10               # any packet involving this IP
  ip.src == 10.0.1.10                # packets FROM this IP
  ip.dst == 10.0.1.10                # packets TO this IP
  ip.addr == 10.0.0.0/8              # any packet in subnet

FILTER BY PORT
  tcp.port == 443
  tcp.dstport == 80
  udp.port == 53

FILTER BY PROTOCOL (type directly in filter bar)
  http        dns         ssl         tls
  ssh         ftp         smtp        imap
  smb         ldap        kerberos    dhcp
  arp         icmp        ntp         rdp

FILTER COMBINED
  ip.addr == 10.0.1.10 and tcp.port == 443
  ip.src == 10.0.1.5 or ip.src == 10.0.1.6
  not arp and not icmp

FOLLOW A TCP STREAM (right-click any packet → Follow → TCP Stream)
  # This reconstructs the full conversation — great for HTTP, SMTP debugging

FIND RETRANSMISSIONS (network performance issue)
  tcp.analysis.retransmission

FIND RESETS (connection refused / abrupt disconnects)
  tcp.flags.reset == 1

FIND SYN PACKETS (new connection attempts)
  tcp.flags.syn == 1 and tcp.flags.ack == 0

FIND PACKETS WITH ERRORS
  tcp.analysis.flags

HTTP REQUESTS AND RESPONSES
  http.request                       # all HTTP requests
  http.response.code == 200          # successful responses
  http.response.code == 500          # server errors
  http contains "password"           # WARNING: plaintext only

DNS QUERIES
  dns.qry.name contains "example"
  dns.flags.rcode != 0               # DNS errors (NXDOMAIN, SERVFAIL etc.)

TLS/SSL
  tls.handshake                      # TLS handshake packets
  ssl.alert_message                  # TLS alerts (certificate errors, etc.)

SECTION 4: NETWORK DIAGNOSTIC SCENARIOS
================================================================================

SCENARIO: "Users can't reach website — is traffic arriving at the server?"
  # On the web server:
  sudo tcpdump -i ens3 -nn -c 50 dst port 80 or dst port 443
  # If no packets: traffic isn't reaching server (check firewall, routing)
  # If packets arrive: server receiving traffic but not responding (check service)

SCENARIO: "Intermittent connection drops — are there TCP resets?"
  sudo tcpdump -i ens3 -nn 'tcp[tcpflags] & tcp-rst != 0' -c 50 -w /tmp/resets.pcap
  # Open in Wireshark, check who is sending RSTs (client or server side)

SCENARIO: "Database connections failing — is port reachable?"
  sudo tcpdump -i any -nn dst port 5432 -c 20
  # From app server, also try: nc -zv db01.example.com 5432

SCENARIO: "DNS lookups failing — is the DNS server responding?"
  sudo tcpdump -i any -nn port 53 -c 20
  # Look for queries going out (src = server) and responses coming back (src = DNS)
  # If no response: DNS server unreachable or dropping UDP 53

SCENARIO: "AD authentication slow — is Kerberos working?"
  sudo tcpdump -i any -nn port 88 -c 50
  # Should see KRB5 requests to DC (port 88) and responses

SCENARIO: "Suspicious traffic — what is this server talking to?"
  sudo tcpdump -i ens3 -nn -c 500 not src host 127.0.0.1 | \
    awk '{print $3}' | cut -d. -f1-4 | sort | uniq -c | sort -rn | head -20
  # Shows all destination IPs and frequency — investigate any unexpected external IPs

SCENARIO: "IP conflict suspected — two hosts with same IP?"
  sudo tcpdump -i ens3 -nn arp | grep "Reply"
  # Multiple MAC addresses responding to the same IP = conflict
  sudo arping -D -I ens3 10.0.1.10 -c 5
  # If exit code 1: duplicate IP detected


================================================================================
SECTION 5: BANDWIDTH AND PERFORMANCE MONITORING
================================================================================

INSTALL TOOLS
  sudo apt install -y iftop nethogs bmon nload

IFTOP — BANDWIDTH BY CONNECTION (like top, for network)
  sudo iftop -i ens3 -n      # -n = no hostname resolution

NETHOGS — BANDWIDTH BY PROCESS
  sudo nethogs ens3

BMON — BANDWIDTH GRAPHING PER INTERFACE
  bmon

NLOAD — REAL-TIME BANDWIDTH METER
  nload ens3

WATCH INTERFACE STATISTICS (built-in, no install)
  watch -n 1 'cat /proc/net/dev | grep ens3'
  # Or:
  watch -n 1 'ip -s link show ens3'

MEASURE ACTUAL THROUGHPUT TO BACKUP SERVER (iperf)
  # On server (receiver):
  sudo apt install -y iperf3
  iperf3 -s

  # On client (sender):
  iperf3 -c 10.0.1.50 -t 10   # 10-second test
  iperf3 -c 10.0.1.50 -t 30 -P 4   # 4 parallel streams
