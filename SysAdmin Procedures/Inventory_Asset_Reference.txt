CHECK EXPIRY OF ALL CERTS (run monthly):
  # Remote:
  for domain in app.example.com internal.example.com; do
    echo -n "$domain: "
    echo | openssl s_client -connect $domain:443 2>/dev/null | openssl x509 -noout -enddate 2>/dev/null
  done

  # Local files:
  find /etc/ssl/certs /etc/letsencrypt/live -name "*.crt" -o -name "cert.pem" 2>/dev/null | while read cert; do
    expiry=$(openssl x509 -noout -enddate -in "$cert" 2>/dev/null | cut -d= -f2)
    echo "$cert: $expiry"
  done

ALERT THRESHOLD: Renew any certificate expiring within 30 days.
